cmake_minimum_required(VERSION 3.16)
project(kompakt_speech LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH armv8-a CACHE STRING "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
if(NOT WHISPER_SOURCE)
  message(FATAL_ERROR "Set WHISPER_SOURCE to the pinned whisper.cpp source directory")
endif()
add_subdirectory(${WHISPER_SOURCE} whisper)
add_library(kompakt_speech SHARED speech.cpp)
target_link_libraries(kompakt_speech PRIVATE whisper)
target_link_options(kompakt_speech PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
